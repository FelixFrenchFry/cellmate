cmake_minimum_required(VERSION 3.14)
project(cellmate LANGUAGES CXX)

add_library(cellmate INTERFACE)
add_library(cellmate::cellmate ALIAS cellmate)
target_include_directories(cellmate INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(cellmate INTERFACE cxx_std_14)

# preserve consistent rounding precision across scalar and SIMD builds
if(MSVC)
	if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
		target_compile_options(cellmate INTERFACE /fp:precise /clang:-ffp-contract=off)
	elseif(MSVC_VERSION LESS 1930)
		target_compile_options(cellmate INTERFACE /fp:strict)
	else()
		target_compile_options(cellmate INTERFACE /fp:precise)
	endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(cellmate INTERFACE -ffp-contract=off)
endif()

# select optional SIMD backend (AVX512, AVX256 or regular scalar)
set(CELLMATE_SIMD "AUTO" CACHE STRING "SIMD backend: AUTO, SCALAR, AVX256 or AVX512")
set_property(CACHE CELLMATE_SIMD PROPERTY STRINGS AUTO SCALAR AVX256 AVX512)
if(NOT CELLMATE_SIMD MATCHES "^(AUTO|SCALAR|AVX256|AVX512)$")
	message(FATAL_ERROR "CELLMATE_SIMD must be AUTO, SCALAR, AVX256 or AVX512")
endif()
target_compile_definitions(cellmate INTERFACE CELLMATE_SIMD_${CELLMATE_SIMD})

option(CELLMATE_ENABLE_OPENMP "Enable OpenMP for cellmate consumers" OFF)
if(CELLMATE_ENABLE_OPENMP)
	find_package(OpenMP REQUIRED COMPONENTS CXX)
	target_link_libraries(cellmate INTERFACE OpenMP::OpenMP_CXX)
endif()

option(CELLMATE_BUILD_BENCHMARKS "Build the cellmate performance benchmarks" OFF)
if(CELLMATE_BUILD_BENCHMARKS)
	add_subdirectory(benchmarks)
endif()

option(CELLMATE_BUILD_EXAMPLES "Build the cellmate usage examples" OFF)
if(CELLMATE_BUILD_EXAMPLES)
	add_subdirectory(examples)
endif()

option(CELLMATE_BUILD_TESTS "Build the cellmate correctness tests" OFF)
if(CELLMATE_BUILD_TESTS)
	add_subdirectory(tests)
endif()
